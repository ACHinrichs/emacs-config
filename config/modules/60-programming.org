* Python

Use Python 3 outside Windows, activate virtual environments through =pyvenv=,
and restart the inferior shell before sending a complete buffer so imports are
fresh.  The restart is deliberately not used for regions.

#+begin_src emacs-lisp
;;; -*- lexical-binding: t; -*-
(unless (eq system-type 'windows-nt) (setq python-shell-interpreter "/usr/bin/python3"))
(use-package pyvenv
  :ensure t
  :config
  (pyvenv-mode 1)
  (setq pyvenv-post-activate-hooks
        (list (lambda () (setq python-shell-interpreter
                                (concat pyvenv-virtual-env "bin/python3"))))
        pyvenv-post-deactivate-hooks
        (list (lambda () (setq python-shell-interpreter "python3")))))
(use-package pylint :ensure t)
(defun ahi-python-shell-run ()
  "Restart Python, send the current buffer, and show the shell."
  (interactive)
  (when (get-buffer-process "*Python*")
    (kill-process (get-buffer-process "*Python*")))
  (run-python (python-shell-parse-command) nil 1)
  (python-shell-send-buffer)
  (unless (get-buffer-window "*Python*" t) (python-shell-switch-to-shell)))
(defun ahi-python-shell-run-region ()
  "Send the active region to Python and show its shell."
  (interactive)
  (python-shell-send-region (region-beginning) (region-end))
  (python-shell-switch-to-shell))
(defun ahi-python-send-buffer-with-args (args)
  "Run the current Python buffer with interactive ARGS."
  (interactive "sPython arguments: ")
  (let ((source-buffer (current-buffer)))
    (with-temp-buffer
      (insert "import sys; sys.argv = '''emacs-interactive " args "'''.split()\n")
      (insert-buffer-substring source-buffer)
      (ahi-python-shell-run))))
(with-eval-after-load 'python
  (define-key python-mode-map (kbd "C-c C-c") #'ahi-python-shell-run)
  (define-key python-mode-map (kbd "C-c C-r") #'ahi-python-shell-run-region))
(global-set-key (kbd "C-c C-a") #'ahi-python-send-buffer-with-args)
(add-hook 'python-major-mode-hook (lambda () (setq auto-composition-mode nil)))
#+end_src

* Rust and templates

Rustic delegates formatting to the system Rust tools.  Jinja files use
=jinja2-mode= and receive the programming-buffer layout from the editing module.

#+begin_src emacs-lisp
(use-package rustic
  :ensure t
  :bind (:map rustic-mode-map
              ("M-j" . lsp-ui-imenu) ("M-?" . lsp-find-references)
              ("C-c C-c l" . flycheck-list-errors) ("C-c C-c a" . lsp-execute-code-action)
              ("C-c C-c r" . lsp-rename) ("C-c C-c q" . lsp-workspace-restart)
              ("C-c C-c Q" . lsp-workspace-shutdown) ("C-c C-c s" . lsp-rust-analyzer-status))
  :config
  (setq rustic-format-on-save t rustic-rustfmt-bin "/usr/bin/rustfmt" rustic-cargo-bin "/usr/bin/cargo")
  (add-hook 'rustic-mode-hook
            (lambda () (when buffer-file-name (setq-local buffer-save-without-query t)))))
(use-package jinja2-mode
  :ensure t
  :mode "\\.jinja2\\'"
  :config (add-to-list 'auto-mode-alist '("\\.jinja2\\'" . jinja2-mode)))
#+end_src

* Formatting

=format-all= enables Black for Python, shfmt for Shell, and djlint for HTML and
Jinja.  Install the corresponding external formatter to activate each entry.

#+begin_src emacs-lisp
(use-package format-all
  :ensure t
  :hook ((prog-mode . format-all-mode) (format-all-mode . format-all-ensure-formatter))
  :config (setq-default format-all-formatters
                        '(("Python" (black)) ("Shell" (shfmt "-i" "4" "-ci"))
                          ("HTML" (djlint "." "--reformat")))))
(with-eval-after-load 'format-all
  (define-format-all-formatter djlint
    (:executable "djlint" :install "pip install djlint" :languages "HTML Jinja2"
     :features :format (format-all--buffer-easy executable "--reformat" "-"))))
(use-package multi-line :ensure t :bind ("C-c m" . multi-line))
#+end_src
